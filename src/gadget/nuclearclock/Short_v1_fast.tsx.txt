// The Inventor's Desk No. 004 "The Clock Inside an Atom" (Now episode, Nolan method, 23 Oct 2026): three clocks (sea 1707,
// pocket today, lab 2026) are set up, then paid off fast, converging on one rule; the head/feet clocks open and close it
// (hairpin loop). A tick that speeds up with each era is the audible clock. Facts and sources:
// docs/inventors_desk/scripts/N2_nuclearclock.md. Until the narration is voiced, timing.json is an estimate and the Short is silent.

const cue = makeCues(T as unknown as Timing);
const {at, next} = cue;
const VOICED = !(T as {estimate?: boolean}).estimate;
export const NUCLEARCLOCK_FRAMES = cue.total + 45;

// the audible clock: ticks speed up era by era (1/s at sea, 2/s Harrison, 4/s caesium, 8/s thorium), and frame the hairpin
const ticks = (a: number, b: number, perSec: number) => {
  const out: number[] = []; for (let f = a; f < b; f += Math.round(FPS / perSec)) out.push(f); return out;
};
const TICKS: [number, number][] = [
  ...ticks(0, at("b02"), 1).map((f) => [f, 0.22] as [number, number]),
  ...ticks(at("b04"), next("b05"), 1).map((f) => [f, 0.2] as [number, number]),
  ...ticks(at("b10"), next("b10"), 2).map((f) => [f, 0.18] as [number, number]),
  ...ticks(at("b11"), next("b11"), 4).map((f) => [f, 0.15] as [number, number]),
  ...ticks(at("b13"), next("b13"), 8).map((f) => [f, 0.12] as [number, number]),
  ...ticks(at("b16"), NUCLEARCLOCK_FRAMES, 1).map((f) => [f, 0.22] as [number, number]),
];
const SFX: [number, string, number][] = [
  [at("b05", "four British warships"), "sfx_whoosh_deep_kinoton.wav", 0.3],
  [at("b09", "A clock is only"), "sfx_underline_stroke_zembacraftworks.wav", 0.35],
  [at("b10", "five seconds"), "sfx_pop_conarb13.wav", 0.3],

/** big rolling tick-rate counter (digits settle from left to right) */
const Rate: React.FC<{value: string; label: string; f0: number; y: number}> = ({value, label, f0, y}) => {
  const fr = useCurrentFrame(); const settle = ease(fr, f0, f0 + 24);
  const shown = value.split("").map((ch, i) => (/\d/.test(ch) && i / value.length > settle ? String((Math.floor(fr * 1.7) + i * 3) % 10) : ch)).join("");
  return (
    <div style={{position: "absolute", left: 0, width: W, top: y, textAlign: "center", opacity: lin(fr, f0 - 2, f0 + 6)}}>
      <div style={{fontFamily: HEAD, fontSize: value.length > 14 ? 62 : 84, color: C.accent, letterSpacing: -1}}>{shown}</div>
      <div style={{fontFamily: MONO, fontSize: 26, letterSpacing: 3, color: C.ink2, marginTop: 8, textTransform: "uppercase"}}>{label}</div>
    </div>
  );
};
